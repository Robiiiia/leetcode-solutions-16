multiply strings

class Solution {

    public String multiply(String num1, String num2) {

        // Если хотя бы одно число равно "0", результат умножения будет "0".
        // equals("0") проверяет, совпадает ли строка с текстом "0".
        if (num1.equals("0") || num2.equals("0")) {
            return "0"; // Возвращаем строку "0" и завершаем метод.
        }

        // Получаем длину первой строки, то есть количество цифр первого числа.
        int n = num1.length();

        // Получаем длину второй строки, то есть количество цифр второго числа.
        int m = num2.length();

        // Создаём массив для хранения цифр результата.
        // При умножении числа из n и m цифр результат содержит максимум n + m цифр.
        int[] result = new int[n + m];

        // Начинаем перебирать цифры первого числа справа налево.
        for (int i = n - 1; i >= 0; i--) {

            // Превращаем текущий символ первой строки в настоящую цифру.
            // Вычитание '0' преобразует символ, например '3', в число 3.
            int digit1 = num1.charAt(i) - '0';

            // Начинаем перебирать цифры второго числа справа налево.
            for (int j = m - 1; j >= 0; j--) {

                // Превращаем текущий символ второй строки в настоящую цифру.
                int digit2 = num2.charAt(j) - '0';

                // Находим позицию, в которой будет храниться результат умножения цифр.
                int pos1 = i + j;

                // Следующая позиция справа предназначена для единиц и переноса.
                int pos2 = i + j + 1;

                // Умножаем две цифры и прибавляем то, что уже хранится в pos2.
                int sum = digit1 * digit2 + result[pos2];

                // Записываем цифру единиц в позицию pos2.
                // Например, если sum = 18, записываем 8.
                result[pos2] = sum % 10;

                // Переносим десятки в предыдущую позицию.
                // Например, если sum = 18, перенос будет равен 1.
                result[pos1] += sum / 10;
            }
        }

        // Создаём объект StringBuilder для сборки ответа из цифр массива.
        StringBuilder answer = new StringBuilder();

        // Перебираем все цифры массива от начала до конца.
        for (int digit : result) {

            // Если ответ пока пустой и текущая цифра равна 0,
            // пропускаем этот ноль, чтобы убрать лишние нули в начале.
            if (answer.length() == 0 && digit == 0) {
                continue; // Переходим к следующей цифре массива.
            }

            // Добавляем текущую цифру в конец ответа.
            answer.append(digit);
        }

        // Превращаем StringBuilder в обычную строку и возвращаем результат.
        return answer.toString();
    }
}
